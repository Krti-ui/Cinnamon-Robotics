import math
from cinnamon_robotics import ArmController

# Exact Link dimensions in centimeters
L1, L2, L3 = 11.5, 9.5, 5.0

def solve_inverse_kinematics(x, y, z):
    """Calculates joint angle degrees for cartesian goal"""
    theta1 = math.atan2(y, x)
    r = math.sqrt(x**2 + y**2)
    z_eff = z - L1
    
    d = math.sqrt(r**2 + z_eff**2)
    cos_theta2 = (L2**2 + d**2 - L3**2) / (2 * L2 * d)
    theta2 = math.atan2(z_eff, r) + math.acos(cos_theta2)
    
    return [math.degrees(theta1), math.degrees(theta2)]

arm = ArmController(port="/dev/ttyUSB0", baud=115200)
arm.move_to_cartesian(x=12.0, y=8.5, z=10.0)
